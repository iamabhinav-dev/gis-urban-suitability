\begin{frame}{Problem and approach}
  \small
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Where can new development reasonably go?}
      \begin{itemize}
        \item Classic GIS answer: weighted overlay with \hl{expert (AHP) weights},
              rarely checked against reality.
        \item Ours: \hl{learn from where building actually happened}, and test every
              method against \hl{real growth}.
      \end{itemize}
    \end{column}
    \begin{column}{0.47\textwidth}
      \textbf{Three questions}
      \begin{enumerate}
        \item Which land is \textcolor{built}{\textbf{built-up}},
              \textcolor{forest}{\textbf{forest}} or \textcolor{usable}{\textbf{usable}}?
        \item Given a building at \textbf{A}, where are similar locations \textbf{B}?
        \item Do data-driven scores beat a classic MCDA overlay?
      \end{enumerate}
      \vspace{0.3em}
      All in \textbf{Python}; 2018 data only $\rightarrow$ tested on growth by 2022/23.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Validation design and leakage rules}
  \small
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Two runs}
      \begin{itemize}
        \item \hl{Validation run}: only data up to 2019 (LULC 2018, OSM 2018-01-01)
              $\rightarrow$ tested on growth by 2022/23
        \item \textbf{Final run}: same models on 2023 data $\rightarrow$ the map
      \end{itemize}
      \textbf{Persistent growth}: non-built in 2018 \emph{and} 2019, built in 2022
      \emph{and} 2023\\[0.2em]
      12,777 candidate cells, \hl{1,125 grew} (8.8\,\%)
    \end{column}
    \begin{column}{0.47\textwidth}
      \textbf{Leakage rules}
      \begin{itemize}
        \item no model sees the cell's own land cover
        \item nothing after 2019 feeds a 2018 score
        \item trained models scored \hl{out of fold} (2\,km blocks)
      \end{itemize}
      \textbf{Always shown}: random and distance to built-up (the bar to beat)\\[0.2em]
      \textbf{Metrics}: AUC, lift in the top 10\,\%, TOC, adjacent vs outlying growth
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Building similarity: A $\rightarrow$ B}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \centering\includegraphics[width=\textwidth]{map_similarity_query_school_1019}
    \end{column}
    \begin{column}{0.48\textwidth}
      \small
      \textbf{Profile}: land cover in 250 / 500\,m rings, distances to built-up, roads and
      water, road density, terrain (never the cell itself)
      \begin{itemize}
        \item Query a building $\rightarrow$ top 20 \textcolor{usable}{\textbf{usable}} cells by
              Euclidean distance
        \item Each match explained: closest and most different features
        \item Self-check: a building's own location is always in the top 1\,\%
      \end{itemize}
      \textbf{Aggregate}: mean distance to the 10 nearest building profiles\\
      $\rightarrow$ \hl{AUC 0.773} on growth, without ever seeing a growth label
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Learned models: predicting growth from 2018}
  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \centering\includegraphics[width=\textwidth]{chart_roc}
    \end{column}
    \begin{column}{0.54\textwidth}
      \small
      \textbf{Change RF}: 2018 surroundings of the 355 cells built by 2020/21
      \begin{itemize}
        \item \hl{AUC 0.779}, above distance to built-up (0.763)
        \item In-sample it would claim 0.997 $\rightarrow$ \textbf{out-of-fold} scores are essential
      \end{itemize}
      \textbf{State RF} (built vs non-built): 0.998 on its own task, only 0.702 on growth
      \vspace{0.4em}

      \textbf{Held-out buildings} (2\,km blocks): RF finds unseen building cells with AUC 0.98\\[0.3em]
      Combined with the similarity: \hl{AUC 0.797}
    \end{column}
  \end{columns}
\end{frame}

        \midrule

        \item The overlay does \hl{worse than one of its own criteria}

\begin{frame}{Limitations}
  \small
  \begin{itemize}
    \item Growth here is mostly \textbf{edge expansion}: learned models gain only a little over
          ``next to existing built-up land''.
    \item \textbf{OSM completeness}: road mapping more than doubled since 2018; OSM buildings
          cover $\sim$1\,\% of built cells in the south (2018); almost all untyped.
    \item \textbf{Land-cover errors}: ESRI generalises urban areas and calls scrub forest
          ``rangeland''; growth labels inherit its errors.
    \item \textbf{Past growth $\neq$ good planning}: zoning, flood risk and infrastructure
          capacity are not in the data.
  \end{itemize}
\end{frame}

\begin{frame}{Conclusion}
  \small
  \begin{enumerate}
    \item \textbf{Land classes}: K-Means (k = 7) $\rightarrow$
          \textcolor{built}{\textbf{built-up}} 348, \textcolor{forest}{\textbf{forest}} 29,
          \textcolor{usable}{\textbf{usable}} 138\,km\textsuperscript{2} (2023); 96.6\,\% agreement with ESRI
    \item \textbf{Building similarity}: any building $\rightarrow$ similar usable locations,
          explained; predicts growth (AUC 0.773) without labels
    \item \textbf{Suitability}: the combined score (\hl{AUC 0.797}, lift 3.45) beats distance to
          built-up (0.763) and a classic AHP overlay (0.636)
  \end{enumerate}
  \vspace{0.5em}
  \textbf{Lesson}: honest validation (leakage rules, baselines) matters more than the model:
  most of the signal is ``next to existing built-up land''; growth away from it is not predicted.
\end{frame}
